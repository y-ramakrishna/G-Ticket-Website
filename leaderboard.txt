    <script>
        function loadLeaderboard() {
            let registrations = JSON.parse(localStorage.getItem("eventRegistrations")) || [];

            // Sort by highest points
            registrations.sort((a, b) => b.points - a.points);

            let table = document.getElementById("leaderboard");

            // Clear previous data (except headers)
            table.innerHTML = `
                <tr>
                    <th>Rank</th>
                    <th>Name</th>
                    <th>Roll Number</th>
                    <th>Email</th>
                    <th>Event</th>
                    <th>Points</th>
                </tr>
            `;

            registrations.forEach((user, index) => {
                let row = table.insertRow();
                row.innerHTML = `
                    <td>${index + 1}</td>
                    <td class="highlight">${user.name}</td>
                    <td>${user.rollNumber}</td>
                    <td>${user.email}</td>
                    <td>${user.event}</td>
                    <td>${user.points}</td>
                `;
            });
        }

        window.onload = loadLeaderboard;
    </script>
