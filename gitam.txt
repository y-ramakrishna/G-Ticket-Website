function searchEvents() {
    let query = document.getElementById("searchBar").value.toLowerCase();
    let events = document.querySelectorAll(".event"); // Select all event cards
    let found = false;

    events.forEach(event => {
        let eventName = event.querySelector("h3").textContent.toLowerCase();
        if (eventName.includes(query)) {
            event.style.display = "inline-block"; // ✅ Show matching event
            found = true;
        } else {
            event.style.display = "none"; // ❌ Hide non-matching events
        }
    });

    let message = document.getElementById("noResultsMessage");
    if (!found) {
        if (!message) {
            message = document.createElement("p");
            message.id = "noResultsMessage";
            message.textContent = "No events found!";
            message.style.color = "red";
            message.style.textAlign = "center";
            document.querySelector(".events-section").appendChild(message);
        }
    } else if (message) {
        message.remove();
    }
}
